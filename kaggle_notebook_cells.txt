# [CELL 4 - CODE]
import time
import subprocess
import sys
from pycloudflared import start_cloudflare_tunnel

# 1. Start FastAPI server in background on port 8000
print("🚀 Starting AutoVideo FastAPI Server...")
server_process = subprocess.Popen(
    [sys.executable, "-m", "uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"],
    stdout=subprocess.PIPE,
    stderr=subprocess.PIPE
)

time.sleep(5)

# 2. Launch Cloudflare Public Tunnel
print("🌐 Creating Public Web Tunnel...")
try:
    public_url = start_cloudflare_tunnel(port=8000)
    print("\n" + "="*70)
    print("🎉 AUTOVIDEO WEB UI IS LIVE!")
    print(f"👉 Public Link: {public_url}")
    print("="*70 + "\n")
    print("အထက်ပါ Public Link ကို နှိပ်ပြီး Browser မှတဆင့် Web UI ကို တိုက်ရိုက် သုံးစွဲနိုင်ပါပြီ။")
except Exception as e:
    print(" Tunnel Error:", e)
